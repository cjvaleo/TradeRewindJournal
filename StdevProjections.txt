//@version=6
indicator("Stdev Projections", overlay=true, max_lines_count=500, max_labels_count=500, max_boxes_count=100)

// =============================================================================
// Stdev Projections (15m, Pivot-Anchored)
// Measurement / annotation tool. Two anchored deviation ladders + leg boxes per
// qualifying swing-low pivot. Negative deviations project UP for both legs.
// Repaint-safe: confirmed pivots only (rightBars lag), no request.security.
// Spec: StdevProjections_handoff_v1.md  |  Build plan: execution plan v1
// =============================================================================

// ---------------------------------------------------------------------------
// INPUTS
// ---------------------------------------------------------------------------

// --- Group: Pivots --------------------------------------------------------
string gPivots = "Pivots"
int pivotLookback = input.int(5, "Pivot lookback (left = right bars)", minval=1, group=gPivots, tooltip="Bars on each side for ta.pivotlow / ta.pivothigh. Confirmation lags by this many bars.")
int recentN       = input.int(10, "Recent window (last N qualifying pivots)", minval=1, group=gPivots, tooltip="The lowest still-valid pivot is chosen from the last N qualifying pivots. A pivot leaves the pool when its wick high crosses its distribution -4 level (see Gate > Invalidation level).")

// --- Group: Gate ----------------------------------------------------------
string gGate = "Gate"
float  gateLevel  = input.float(-4.5, "Gate level (L)", step=0.25, group=gGate, tooltip="A pivot qualifies only if the rally stays below this projected level. Default -4.5 = the manipulation-ladder cutoff.")
string gateSource = input.string("Manipulation", "Gate source ladder", options=["Manipulation", "Distribution"], group=gGate, tooltip="Which ladder's anchors price the gate. Default: Manipulation (priorHigh -> pivotLow).")
float  breakLevel = input.float(-4.5, "Invalidation level (L)", step=0.25, group=gGate, tooltip="A qualified pivot is dropped once a bar's wick HIGH crosses this level of its DISTRIBUTION ladder (nextHigh -> pivotLow). Only still-valid pivots are drawn.")

// --- Group: Manipulation Ladder -------------------------------------------
string gManip = "Manipulation Ladder"
bool showManipBox    = input.bool(false, "Show manipulation box",    group=gManip, tooltip="Off for now — only the distribution leg is marked.")
bool showManipLadder = input.bool(false, "Show manipulation ladder", group=gManip, tooltip="Off for now — only the distribution leg is marked.")

// --- Group: Distribution Ladder -------------------------------------------
string gDist = "Distribution Ladder"
bool showDistBox    = input.bool(true, "Show distribution box",    group=gDist)
bool showDistLadder = input.bool(true, "Show distribution ladder", group=gDist)

// --- Group: Levels (per-level toggles) ------------------------------------
string gLevels = "Levels"
bool keyLevelsOnly   = input.bool(false, "Key levels only (overrides individual toggles)", group=gLevels, tooltip="When on, draws only the colored key levels plus 0 and 1.")
int  levelExtendBars = input.int(0, "Level extend bars (past leg end)", minval=0, group=gLevels, tooltip="Each ladder spans only its own leg (start bar -> last bar of that leg). This optionally extends every level line/label this many bars further right. 0 = hug the leg exactly.")
bool lvl_1    = input.bool(true, "1.0",    group=gLevels, inline="L1")
bool lvl_943  = input.bool(true, "0.943",  group=gLevels, inline="L1")
bool lvl_79   = input.bool(true, "0.79",   group=gLevels, inline="L1")
bool lvl_705  = input.bool(true, "0.705",  group=gLevels, inline="L2")
bool lvl_618  = input.bool(true, "0.618",  group=gLevels, inline="L2")
bool lvl_5    = input.bool(true, "0.5",    group=gLevels, inline="L2")
bool lvl_0    = input.bool(true, "0.0",    group=gLevels, inline="L3")
bool lvl_n1   = input.bool(true, "-1.0",   group=gLevels, inline="L3")
bool lvl_n15  = input.bool(true, "-1.5",   group=gLevels, inline="L3")
bool lvl_n2   = input.bool(true, "-2.0",   group=gLevels, inline="L4")
bool lvl_n225 = input.bool(true, "-2.25",  group=gLevels, inline="L4")
bool lvl_n25  = input.bool(true, "-2.5",   group=gLevels, inline="L4")
bool lvl_n4   = input.bool(true, "-4.0",   group=gLevels, inline="L5")
bool lvl_n425 = input.bool(true, "-4.25",  group=gLevels, inline="L5")
bool lvl_n45  = input.bool(true, "-4.5",   group=gLevels, inline="L5")

// --- Group: Colors & Style ------------------------------------------------
string gColors = "Colors & Style"
bool  showLabels    = input.bool(true, "Show level labels", group=gColors)
color manipBoxColor = input.color(color.new(color.purple, 85), "Manipulation box", group=gColors)
color distBoxColor  = input.color(color.new(color.teal,   85), "Distribution box", group=gColors)
color stdLevelColor = input.color(color.new(color.gray,    0), "Standard level",   group=gColors)
color keyLevelColor = input.color(color.orange,                "Key level (0.618/0.705/0.79/-2.25/-2.5)", group=gColors)
color level4Color   = input.color(color.red,                   "Level -4",         group=gColors)
color level45Color  = input.color(color.new(color.maroon, 0),  "Level -4.5 (gate cutoff)", group=gColors)

// --- Group: Alerts --------------------------------------------------------
string gAlerts = "Alerts"
bool enableAlert = input.bool(false, "Alert on new qualifying pivot", group=gAlerts)

// --- Group: Debug ---------------------------------------------------------
string gDebug = "Debug"
bool debugMode = input.bool(false, "DEBUG: show pivot pairing connectors", group=gDebug, tooltip="Purple manip leg + teal/gray dist leg + dashed gate line, proving priorHigh->low->nextHigh pairing and the gate verdict. Diagnostic only.")

// ---------------------------------------------------------------------------
// TYPES
// ---------------------------------------------------------------------------
type Pivot
    int   barIndex          // pivot low bar
    float price             // pivot low (wick: low at pivot bar)
    int   priorHighBar      // most recent confirmed pivot high before the low
    float priorHigh         // wick: high at that bar
    int   nextHighBar       // first confirmed pivot high after the low
    float nextHigh          // wick: high at that bar
    bool  hasNextHigh       // true once the distribution endpoint confirms
    bool  evaluated         // gate evaluated exactly once
    bool  qualified         // passed the gate -> drawable
    array<line>  lines      // handles for cleanup
    array<box>   boxes
    array<label> labels

// ---------------------------------------------------------------------------
// LEVELS + SHARED HELPERS
// ---------------------------------------------------------------------------
// Master ladder levels (negative L projects UP — above anchor0).
var array<float> levels = array.from(1.0, 0.943, 0.79, 0.705, 0.618, 0.5, 0.0, -1.0, -1.5, -2.0, -2.25, -2.5, -4.0, -4.25, -4.5)
// Per-level enable flags, aligned 1:1 with `levels`.
var array<bool>  levelEnabled = array.from(lvl_1, lvl_943, lvl_79, lvl_705, lvl_618, lvl_5, lvl_0, lvl_n1, lvl_n15, lvl_n2, lvl_n225, lvl_n25, lvl_n4, lvl_n425, lvl_n45)

// Level price: L=0 -> anchor0, L=1 -> anchor1, negative L -> above anchor0 (up).
levelPrice(float anchor0, float anchor1, float L) => anchor0 + L * (anchor1 - anchor0)

// Key (orange) levels echo the user's fib config.
isKeyOrange(float L) => L == 0.618 or L == 0.705 or L == 0.79 or L == -2.25 or L == -2.5

// Draw one ladder for pivot `p`: a line per enabled level priced off (a0,a1),
// from leftBar to rightBar, with a right-edge label. Handles pushed into
// p.lines / p.labels for group cleanup. Respects keyLevelsOnly + per-level toggles.
drawLadder(Pivot p, float a0, float a1, int leftBar, int rightBar) =>
    for li = 0 to array.size(levels) - 1
        float L     = array.get(levels, li)
        bool  keyO  = isKeyOrange(L)
        bool  isRed = L == -4.0
        bool  is45  = L == -4.5
        bool  colored = keyO or isRed or is45
        // keyLevelsOnly overrides individual toggles: colored levels + 0/1 only.
        bool  enabled = keyLevelsOnly ? (colored or L == 0.0 or L == 1.0) : array.get(levelEnabled, li)
        if enabled
            float pr   = levelPrice(a0, a1, L)
            color lcol = is45 ? level45Color : isRed ? level4Color : keyO ? keyLevelColor : stdLevelColor
            ls = is45 ? line.style_dashed : line.style_solid
            ln = line.new(leftBar, pr, rightBar, pr, xloc=xloc.bar_index, color=lcol, width=1, style=ls)
            array.push(p.lines, ln)
            if showLabels
                lb = label.new(rightBar, pr, str.tostring(L), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(color.black, 100), textcolor=lcol, size=size.tiny)
                array.push(p.labels, lb)
    array.size(p.lines)

// Delete all drawn objects of `p` and clear its handle arrays (safe to re-render after).
clearPivot(Pivot p) =>
    for ln in p.lines
        line.delete(ln)
    for lb in p.labels
        label.delete(lb)
    for bx in p.boxes
        box.delete(bx)
    p.lines.clear()
    p.labels.clear()
    p.boxes.clear()

// Draw the two boxes + both ladders for a qualified pivot; handles stored on `p`.
// Each ladder spans only its own leg (start bar -> last bar of that leg),
// optionally extended right by levelExtendBars (default 0).
renderPivot(Pivot p) =>
    if showManipBox
        array.push(p.boxes, box.new(p.priorHighBar, p.priorHigh, p.barIndex, p.price, xloc=xloc.bar_index, border_color=color.new(color.purple, 40), border_width=1, bgcolor=manipBoxColor))
    if showDistBox
        array.push(p.boxes, box.new(p.barIndex, p.nextHigh, p.nextHighBar, p.price, xloc=xloc.bar_index, border_color=color.new(color.teal, 40), border_width=1, bgcolor=distBoxColor))
    if showManipLadder
        drawLadder(p, p.priorHigh, p.price, p.priorHighBar, p.barIndex + levelExtendBars)      // manip leg: priorHigh -> low
    if showDistLadder
        drawLadder(p, p.nextHigh, p.price, p.barIndex, p.nextHighBar + levelExtendBars)        // dist  leg: low -> nextHigh
    array.size(p.boxes)

// ---------------------------------------------------------------------------
// STATE
// ---------------------------------------------------------------------------
var array<Pivot> pivots = array.new<Pivot>()   // PENDING lows only (removed once evaluated)
var array<Pivot> valid  = array.new<Pivot>()   // qualified & not-yet-invalidated candidates (selection pool)
var array<Pivot> drawn  = array.new<Pivot>()   // currently rendered pivots (1-2): most-recent-valid + lowest-valid
var int          drawnRecentBar = -1           // barIndex of the drawn most-recent pivot (-1 = none)
var int          drawnLowBar    = -1           // barIndex of the drawn lowest pivot (-1 = none)
var array<line>  dbgLines = array.new<line>()   // debug connectors

var int   lastPHBar   = na   // most recent confirmed pivot-high bar (the prior swing high)
var float lastPHPrice = na   // its wick high

// ta.* computed every bar at top level — never gated on data-dependent branches.
int   pivotBar = bar_index - pivotLookback
float ph = ta.pivothigh(pivotLookback, pivotLookback)
float pl = ta.pivotlow(pivotLookback, pivotLookback)

// ---------------------------------------------------------------------------
// TIMEFRAME GUARD (15m only) + non-15m note
// ---------------------------------------------------------------------------
bool is15m = timeframe.in_seconds() == 900

var label tfNote = na
if barstate.islast
    if not is15m
        if na(tfNote)
            tfNote := label.new(bar_index, high, "Stdev Projections — switch to 15m", xloc=xloc.bar_index, style=label.style_label_left, color=color.new(color.gray, 20), textcolor=color.white, size=size.normal)
        else
            label.set_xy(tfNote, bar_index, high)
    else if not na(tfNote)
        label.delete(tfNote)
        tfNote := na

// =============================================================================
// MAIN — runs only on the 15m chart
// =============================================================================
if is15m
    // Per-bar flags for the qualifying alert (Module B owns the alert).
    bool  newQualify = false
    float qLowPx     = na
    float qNextPx    = na

    // -------------------------------------------------------------------------
    // MODULE A (high) + MODULE B (gate) + MODULE C (render)
    // A confirmed pivot HIGH is the only event that can complete a pending low
    // (set its nextHigh), so gate evaluation + rendering are guarded under it —
    // O(pending) work, only on high-confirm bars. Iterate backwards so we can
    // remove completed records in-place.
    // -------------------------------------------------------------------------
    if not na(ph)
        if array.size(pivots) > 0
            for i = array.size(pivots) - 1 to 0
                Pivot p = array.get(pivots, i)
                if (not p.hasNextHigh) and p.barIndex < pivotBar
                    // -- Module A: distribution endpoint (first high after the low)
                    p.nextHighBar := pivotBar
                    p.nextHigh    := ph
                    p.hasNextHigh := true

                    // -- Module B: one-shot gate evaluation
                    float gA0    = gateSource == "Manipulation" ? p.priorHigh : p.nextHigh
                    float gPrice = levelPrice(gA0, p.price, gateLevel)
                    p.qualified := p.nextHigh < gPrice   // rally stayed below the gate -> contained reversal
                    p.evaluated := true

                    // -- Debug connectors (manip leg, dist leg colored by verdict, gate line)
                    if debugMode
                        color dcol = p.qualified ? color.new(color.teal, 0) : color.new(color.gray, 40)
                        array.push(dbgLines, line.new(p.priorHighBar, p.priorHigh, p.barIndex, p.price, xloc=xloc.bar_index, color=color.new(color.purple, 0), width=1))
                        array.push(dbgLines, line.new(p.barIndex, p.price, p.nextHighBar, p.nextHigh, xloc=xloc.bar_index, color=dcol, width=1))
                        int gLeft = gateSource == "Manipulation" ? p.priorHighBar : p.barIndex
                        array.push(dbgLines, line.new(gLeft, gPrice, p.nextHighBar, gPrice, xloc=xloc.bar_index, color=color.new(color.maroon, 0), width=1, style=line.style_dashed))

                    // -- Module C (1/3): qualified pivots enter the live candidate pool
                    if p.qualified
                        newQualify := true
                        qLowPx     := p.price
                        qNextPx    := p.nextHigh
                        array.push(valid, p)

                    // -- evaluated -> no longer pending; drop from the pending list
                    array.remove(pivots, i)
        // record this high as the prior swing high for future lows
        lastPHBar   := pivotBar
        lastPHPrice := ph

    // -------------------------------------------------------------------------
    // MODULE C (2/3): invalidation sweep + pool bound (runs every bar).
    //   Drop any candidate whose wick HIGH has crossed its distribution -4 level
    //   (anchor0 = nextHigh, anchor1 = pivotLow). Then bound the pool to the last
    //   recentN qualifiers so 'lowest' is chosen from a recent set.
    // -------------------------------------------------------------------------
    if array.size(valid) > 0
        for i = array.size(valid) - 1 to 0
            Pivot p = array.get(valid, i)
            float breakPx = levelPrice(p.nextHigh, p.price, breakLevel)   // distribution-ladder break level
            if high > breakPx
                array.remove(valid, i)
    while array.size(valid) > recentN
        array.shift(valid)

    // -------------------------------------------------------------------------
    // MODULE C (3/3): draw the most-recent-valid + lowest-valid candidates
    //   (distribution leg only). Both come from `valid`, so both are unbroken.
    //   Redraw only when the chosen pair changes; otherwise keep their lines
    //   extended to the current bar so the levels stay actionable live.
    // -------------------------------------------------------------------------
    // Target pair, identified by barIndex (-1 = none). valid is chronological,
    // so the last element is the most-recent-valid pivot.
    int tRecentBar = -1
    int tLowBar    = -1
    if array.size(valid) > 0
        tRecentBar := array.get(valid, array.size(valid) - 1).barIndex
        Pivot lo = array.get(valid, 0)
        for p in valid
            if p.price < lo.price
                lo := p
        tLowBar := lo.barIndex

    if tRecentBar != drawnRecentBar or tLowBar != drawnLowBar
        // wipe current draw set, then render the new pair (dedup if identical)
        for d in drawn
            clearPivot(d)
        drawn.clear()
        if array.size(valid) > 0
            Pivot pRecent = array.get(valid, array.size(valid) - 1)
            Pivot pLow    = array.get(valid, 0)
            for p in valid
                if p.price < pLow.price
                    pLow := p
            renderPivot(pLow)
            array.push(drawn, pLow)
            if pRecent.barIndex != pLow.barIndex
                renderPivot(pRecent)
                array.push(drawn, pRecent)
        drawnRecentBar := tRecentBar
        drawnLowBar    := tLowBar

    // keep drawn ladders live: extend lines/labels to the current bar
    for d in drawn
        for ln in d.lines
            line.set_x2(ln, bar_index)
        for lb in d.labels
            label.set_x(lb, bar_index)

    // -------------------------------------------------------------------------
    // MODULE A (low): create a pending record anchored to the prior swing high.
    // Skip if no prior high exists yet (start of history) — never half-anchored.
    // -------------------------------------------------------------------------
    if not na(pl)
        if not na(lastPHBar)
            array.push(pivots, Pivot.new(barIndex = pivotBar, price = pl, priorHighBar = lastPHBar, priorHigh = lastPHPrice, nextHighBar = na, nextHigh = na, hasNextHigh = false, evaluated = false, qualified = false, lines = array.new<line>(), boxes = array.new<box>(), labels = array.new<label>()))

    // Safety cap on pending records (no handles to free — pending = undrawn).
    while array.size(pivots) > 100
        array.shift(pivots)

    // Keep debug connectors bounded.
    if debugMode
        while array.size(dbgLines) > 60
            line.delete(array.shift(dbgLines))

    // -------------------------------------------------------------------------
    // ALERTS — fires once per pivot on the qualifying transition (off by default)
    // -------------------------------------------------------------------------
    if enableAlert and newQualify
        alert("Stdev Projections: new qualifying pivot — low " + str.tostring(qLowPx) + ", reversal high " + str.tostring(qNextPx), alert.freq_once_per_bar_close)
