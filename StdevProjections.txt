//@version=6
indicator("Stdev Projections", overlay=true, max_lines_count=500, max_labels_count=500, max_boxes_count=100)

// =============================================================================
// Stdev Projections (15m, Pivot-Anchored) — bullish + bearish distribution legs
//
// Bullish: anchor a swing LOW; distribution leg = low -> next high (rally);
//   negative deviations project UP. Draws the lowest-valid + most-recent-valid.
// Bearish: anchor a swing HIGH; distribution leg = high -> next low (selloff);
//   negative deviations project DOWN. Draws the highest-valid + most-recent-valid.
//
// A pivot QUALIFIES if its distribution endpoint stayed inside the gate (-4.5),
// and stays VALID until a later wick crosses its distribution -4.5 level.
// Repaint-safe: confirmed pivots only (lookback lag), no request.security.
// =============================================================================

// ---------------------------------------------------------------------------
// INPUTS
// ---------------------------------------------------------------------------

// --- Group: Pivots --------------------------------------------------------
string gPivots = "Pivots"
int pivotLookback = input.int(5, "Pivot lookback (left = right bars)", minval=1, group=gPivots, tooltip="Bars on each side for ta.pivotlow / ta.pivothigh. Confirmation lags by this many bars.")
int recentN       = input.int(10, "Recent window (last N qualifying pivots)", minval=1, group=gPivots, tooltip="Per direction, the extreme (lowest/highest) still-valid pivot is chosen from the last N qualifying pivots.")

// --- Group: Gate ----------------------------------------------------------
string gGate = "Gate"
float  gateLevel  = input.float(-4.5, "Gate level (L)", step=0.25, group=gGate, tooltip="A pivot qualifies only if its distribution endpoint stayed inside this projected level.")
string gateSource = input.string("Manipulation", "Gate source ladder", options=["Manipulation", "Distribution"], group=gGate, tooltip="Which ladder's anchors price the gate. Default: Manipulation.")
float  breakLevel = input.float(-4.5, "Invalidation level (L)", step=0.25, group=gGate, tooltip="A qualified pivot is dropped once a wick crosses this level of its DISTRIBUTION ladder (high for bull, low for bear).")

// --- Group: Bullish (up projections) --------------------------------------
string gBull = "Bullish (up projections)"
bool showBullDistBox    = input.bool(true, "Show bullish box",    group=gBull)
bool showBullDistLadder = input.bool(true, "Show bullish ladder", group=gBull)

// --- Group: Bearish (down projections) ------------------------------------
string gBear = "Bearish (down projections)"
bool showBearDistBox    = input.bool(true, "Show bearish box",    group=gBear)
bool showBearDistLadder = input.bool(true, "Show bearish ladder", group=gBear)

// --- Group: Manipulation Ladder (advanced) --------------------------------
string gManip = "Manipulation Ladder (advanced)"
bool showManipBox    = input.bool(false, "Show manipulation box",    group=gManip, tooltip="Off by default — only distribution legs are marked.")
bool showManipLadder = input.bool(false, "Show manipulation ladder", group=gManip, tooltip="Off by default — only distribution legs are marked.")

// --- Group: Levels (per-level toggles) ------------------------------------
string gLevels = "Levels"
bool keyLevelsOnly   = input.bool(false, "Key levels only (overrides individual toggles)", group=gLevels, tooltip="When on, draws only the colored key levels plus 0 and 1.")
int  levelExtendBars = input.int(0, "Level extend bars (past leg end)", minval=0, group=gLevels, tooltip="Each ladder spans only its marked leg (first marked candle -> last marked candle). This optionally extends every line/label this many bars further right. 0 = hug the leg exactly.")
bool lvl_1    = input.bool(true, "1.0",    group=gLevels, inline="L1")
bool lvl_943  = input.bool(true, "0.943",  group=gLevels, inline="L1")
bool lvl_79   = input.bool(true, "0.79",   group=gLevels, inline="L1")
bool lvl_705  = input.bool(true, "0.705",  group=gLevels, inline="L2")
bool lvl_618  = input.bool(true, "0.618",  group=gLevels, inline="L2")
bool lvl_5    = input.bool(true, "0.5",    group=gLevels, inline="L2")
bool lvl_0    = input.bool(true, "0.0",    group=gLevels, inline="L3")
bool lvl_n1   = input.bool(true, "-1.0",   group=gLevels, inline="L3")
bool lvl_n15  = input.bool(true, "-1.5",   group=gLevels, inline="L3")
bool lvl_n2   = input.bool(true, "-2.0",   group=gLevels, inline="L4")
bool lvl_n225 = input.bool(true, "-2.25",  group=gLevels, inline="L4")
bool lvl_n25  = input.bool(true, "-2.5",   group=gLevels, inline="L4")
bool lvl_n4   = input.bool(true, "-4.0",   group=gLevels, inline="L5")
bool lvl_n425 = input.bool(true, "-4.25",  group=gLevels, inline="L5")
bool lvl_n45  = input.bool(true, "-4.5",   group=gLevels, inline="L5")

// --- Group: Colors & Style ------------------------------------------------
string gColors = "Colors & Style"
bool  showLabels    = input.bool(true, "Show level labels", group=gColors)
color distBoxColor  = input.color(color.new(color.teal,   85), "Bullish distribution box", group=gColors)
color bearBoxColor  = input.color(color.new(color.red,    85), "Bearish distribution box", group=gColors)
color manipBoxColor = input.color(color.new(color.purple, 85), "Manipulation box", group=gColors)
color stdLevelColor = input.color(color.new(color.gray,    0), "Standard level",   group=gColors)
color keyLevelColor = input.color(color.orange,                "Key level (0.618/0.705/0.79/-2.25/-2.5)", group=gColors)
color level4Color   = input.color(color.red,                   "Level -4",         group=gColors)
color level45Color  = input.color(color.new(color.maroon, 0),  "Level -4.5 (gate cutoff)", group=gColors)

// --- Group: Alerts --------------------------------------------------------
string gAlerts = "Alerts"
bool enableAlert = input.bool(false, "Alert on new qualifying pivot", group=gAlerts)

// ---------------------------------------------------------------------------
// TYPES  (direction-neutral: same fields serve bull & bear)
// ---------------------------------------------------------------------------
type Pivot
    bool  bear
    int   barIndex
    float price
    int   mBar
    float mPrice
    int   dBar
    float dPrice
    bool  hasDist
    bool  evaluated
    bool  qualified
    array<line>  lines
    array<box>   boxes
    array<label> labels

// ---------------------------------------------------------------------------
// LEVELS + SHARED HELPERS
// ---------------------------------------------------------------------------
var array<float> levels = array.from(1.0, 0.943, 0.79, 0.705, 0.618, 0.5, 0.0, -1.0, -1.5, -2.0, -2.25, -2.5, -4.0, -4.25, -4.5)
var array<bool>  levelEnabled = array.from(lvl_1, lvl_943, lvl_79, lvl_705, lvl_618, lvl_5, lvl_0, lvl_n1, lvl_n15, lvl_n2, lvl_n225, lvl_n25, lvl_n4, lvl_n425, lvl_n45)

levelPrice(float anchor0, float anchor1, float L) => anchor0 + L * (anchor1 - anchor0)

isKeyOrange(float L) => L == 0.618 or L == 0.705 or L == 0.79 or L == -2.25 or L == -2.5

drawLadder(Pivot p, float a0, float a1, int leftBar, int rightBar) =>
    for li = 0 to array.size(levels) - 1
        float L = array.get(levels, li)
        bool keyO = isKeyOrange(L)
        bool isRed = L == -4.0
        bool is45 = L == -4.5
        bool colored = keyO or isRed or is45
        bool enabled = keyLevelsOnly ? (colored or L == 0.0 or L == 1.0) : array.get(levelEnabled, li)
        if enabled
            float pr = levelPrice(a0, a1, L)
            color lcol = is45 ? level45Color : isRed ? level4Color : keyO ? keyLevelColor : stdLevelColor
            ls = is45 ? line.style_dashed : line.style_solid
            ln = line.new(leftBar, pr, rightBar, pr, xloc=xloc.bar_index, color=lcol, width=1, style=ls)
            array.push(p.lines, ln)
            if showLabels
                lb = label.new(rightBar, pr, str.tostring(L), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(color.black, 100), textcolor=lcol, size=size.tiny)
                array.push(p.labels, lb)
    array.size(p.lines)

clearPivot(Pivot p) =>
    for ln in p.lines
        line.delete(ln)
    for lb in p.labels
        label.delete(lb)
    for bx in p.boxes
        box.delete(bx)
    p.lines.clear()
    p.labels.clear()
    p.boxes.clear()

renderPivot(Pivot p) =>
    bool showBox = p.bear ? showBearDistBox : showBullDistBox
    bool showLad = p.bear ? showBearDistLadder : showBullDistLadder
    color boxBg = p.bear ? bearBoxColor : distBoxColor
    color boxBd = p.bear ? color.new(color.red, 40) : color.new(color.teal, 40)
    if showManipBox
        array.push(p.boxes, box.new(p.mBar, math.max(p.mPrice, p.price), p.barIndex, math.min(p.mPrice, p.price), xloc=xloc.bar_index, border_color=color.new(color.purple, 40), border_width=1, bgcolor=manipBoxColor))
    if showManipLadder
        drawLadder(p, p.mPrice, p.price, p.mBar, p.barIndex + levelExtendBars)
    if showBox
        array.push(p.boxes, box.new(p.barIndex, math.max(p.dPrice, p.price), p.dBar, math.min(p.dPrice, p.price), xloc=xloc.bar_index, border_color=boxBd, border_width=1, bgcolor=boxBg))
    if showLad
        drawLadder(p, p.dPrice, p.price, p.barIndex, p.dBar + levelExtendBars)
    array.size(p.boxes)

completePending(array<Pivot> pending, array<Pivot> validArr, int endBar, float endPrice, bool bear) =>
    if array.size(pending) > 0
        for i = array.size(pending) - 1 to 0
            Pivot p = array.get(pending, i)
            if (not p.hasDist) and p.barIndex < endBar
                p.dBar    := endBar
                p.dPrice  := endPrice
                p.hasDist := true
                float gA0 = gateSource == "Manipulation" ? p.mPrice : p.dPrice
                float gP = levelPrice(gA0, p.price, gateLevel)
                p.qualified := bear ? (p.dPrice > gP) : (p.dPrice < gP)
                p.evaluated := true
                if p.qualified
                    array.push(validArr, p)
                    if enableAlert
                        alert("Stdev Projections: new qualifying " + (bear ? "bearish" : "bullish") + " pivot @ " + str.tostring(p.price), alert.freq_once_per_bar_close)
                array.remove(pending, i)
    array.size(validArr)

invalidate(array<Pivot> pool, bool bear) =>
    if array.size(pool) > 0
        for i = array.size(pool) - 1 to 0
            Pivot p = array.get(pool, i)
            float breakPx = levelPrice(p.dPrice, p.price, breakLevel)
            bool broken = bear ? (low < breakPx) : (high > breakPx)
            if broken
                array.remove(pool, i)
    while array.size(pool) > recentN
        array.shift(pool)
    array.size(pool)

updateDrawn(array<Pivot> pool, array<Pivot> drawnArr, array<int> st, bool bear) =>
    int n = array.size(pool)
    int tRecentBar = -1
    int tExtBar = -1
    if n > 0
        tRecentBar := array.get(pool, n - 1).barIndex
        Pivot ext = array.get(pool, 0)
        for p in pool
            if (bear ? (p.price > ext.price) : (p.price < ext.price))
                ext := p
        tExtBar := ext.barIndex
    if tRecentBar != array.get(st, 0) or tExtBar != array.get(st, 1)
        for d in drawnArr
            clearPivot(d)
        drawnArr.clear()
        if n > 0
            Pivot pRecent = array.get(pool, n - 1)
            Pivot pExt = array.get(pool, 0)
            for p in pool
                if (bear ? (p.price > pExt.price) : (p.price < pExt.price))
                    pExt := p
            renderPivot(pExt)
            array.push(drawnArr, pExt)
            if pRecent.barIndex != pExt.barIndex
                renderPivot(pRecent)
                array.push(drawnArr, pRecent)
        array.set(st, 0, tRecentBar)
        array.set(st, 1, tExtBar)
    // NOTE: no live-extend — each ladder spans only its marked leg (first
    // marked candle -> last marked candle), set once in renderPivot.
    array.size(drawnArr)

// ---------------------------------------------------------------------------
// STATE
// ---------------------------------------------------------------------------
var array<Pivot> bullPending = array.new<Pivot>()
var array<Pivot> bearPending = array.new<Pivot>()
var array<Pivot> bullValid   = array.new<Pivot>()
var array<Pivot> bearValid   = array.new<Pivot>()
var array<Pivot> bullDrawn   = array.new<Pivot>()
var array<Pivot> bearDrawn   = array.new<Pivot>()
var array<int>   bullSt = array.from(-1, -1)
var array<int>   bearSt = array.from(-1, -1)

var int   lastPHBar   = na
var float lastPHPrice = na
var int   lastPLBar   = na
var float lastPLPrice = na

int   pivotBar = bar_index - pivotLookback
float ph = ta.pivothigh(pivotLookback, pivotLookback)
float pl = ta.pivotlow(pivotLookback, pivotLookback)

// ---------------------------------------------------------------------------
// TIMEFRAME GUARD (15m only) + non-15m note
// ---------------------------------------------------------------------------
bool is15m = timeframe.in_seconds() == 900

var label tfNote = na
if barstate.islast
    if not is15m
        if na(tfNote)
            tfNote := label.new(bar_index, high, "Stdev Projections — switch to 15m", xloc=xloc.bar_index, style=label.style_label_left, color=color.new(color.gray, 20), textcolor=color.white, size=size.normal)
        else
            label.set_xy(tfNote, bar_index, high)
    else if not na(tfNote)
        label.delete(tfNote)
        tfNote := na

// =============================================================================
// MAIN — runs only on the 15m chart
// =============================================================================
if is15m
    // confirmed pivot HIGH: completes pending BULL lows (their nextHigh);
    // starts a new pending BEAR high anchored to the prior low.
    if not na(ph)
        completePending(bullPending, bullValid, pivotBar, ph, false)
        if not na(lastPLBar)
            array.push(bearPending, Pivot.new(bear = true, barIndex = pivotBar, price = ph, mBar = lastPLBar, mPrice = lastPLPrice, dBar = na, dPrice = na, hasDist = false, evaluated = false, qualified = false, lines = array.new<line>(), boxes = array.new<box>(), labels = array.new<label>()))
        lastPHBar := pivotBar
        lastPHPrice := ph

    // confirmed pivot LOW: completes pending BEAR highs (their nextLow);
    // starts a new pending BULL low anchored to the prior high.
    if not na(pl)
        completePending(bearPending, bearValid, pivotBar, pl, true)
        if not na(lastPHBar)
            array.push(bullPending, Pivot.new(bear = false, barIndex = pivotBar, price = pl, mBar = lastPHBar, mPrice = lastPHPrice, dBar = na, dPrice = na, hasDist = false, evaluated = false, qualified = false, lines = array.new<line>(), boxes = array.new<box>(), labels = array.new<label>()))
        lastPLBar := pivotBar
        lastPLPrice := pl

    while array.size(bullPending) > 100
        array.shift(bullPending)
    while array.size(bearPending) > 100
        array.shift(bearPending)

    invalidate(bullValid, false)
    invalidate(bearValid, true)

    updateDrawn(bullValid, bullDrawn, bullSt, false)
    updateDrawn(bearValid, bearDrawn, bearSt, true)
